cmake_minimum_required(VERSION 3.16)
project(route_incident_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(tcp_connect_probe cpp/tcp_connect_probe.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(tcp_connect_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(epoll_connect_sampler cpp/epoll_connect_sampler.cpp)
  add_executable(rtnetlink_monitor cpp/rtnetlink_monitor.cpp)
  add_executable(udp_feed_monitor cpp/udp_feed_monitor.cpp)
  add_executable(udp_pipeline cpp/udp_pipeline.cpp)
  add_executable(udp_publisher cpp/udp_publisher.cpp)
  add_executable(spsc_ring_test tests/native/test_spsc_ring.cpp)
  find_package(Threads REQUIRED)
  target_link_libraries(udp_pipeline PRIVATE Threads::Threads)
  target_link_libraries(spsc_ring_test PRIVATE Threads::Threads)
  enable_testing()
  add_test(NAME spsc_concurrent_sequence_and_boundedness COMMAND spsc_ring_test)
  add_executable(connect_reactor_compare cpp/connect_reactor_compare.cpp)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(epoll_connect_sampler PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(rtnetlink_monitor PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(udp_feed_monitor PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(udp_pipeline PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(udp_publisher PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(spsc_ring_test PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(connect_reactor_compare PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
